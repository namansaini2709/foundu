CREATE TABLE `foundu_submissions` (
  `id` int NOT NULL AUTO_INCREMENT,
  `job_id` int NOT NULL,
  `freelancer_id` int NOT NULL,
  `founder_id` int NOT NULL,
  `instructions` text,
  `expectations` text,
  `deadline` datetime DEFAULT NULL,
  `founder_files` json DEFAULT NULL,
  `freelancer_work_desc` text,
  `freelancer_work_files` json DEFAULT NULL,
  `freelancer_work_links` json DEFAULT NULL,
  `status` enum('pending_setup','in_progress','submitted','completed','changes_requested') DEFAULT 'pending_setup',
  `feedback` text,
  `progress` int DEFAULT '0',
  `created_at` timestamp NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at` timestamp NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `unique_job_freelancer` (`job_id`,`freelancer_id`),
  KEY `freelancer_id` (`freelancer_id`),
  KEY `founder_id` (`founder_id`),
  CONSTRAINT `foundu_submissions_ibfk_1` FOREIGN KEY (`job_id`) REFERENCES `foundu_jobs` (`job_id`) ON DELETE CASCADE,
  CONSTRAINT `foundu_submissions_ibfk_2` FOREIGN KEY (`freelancer_id`) REFERENCES `users` (`id`),
  CONSTRAINT `foundu_submissions_ibfk_3` FOREIGN KEY (`founder_id`) REFERENCES `users` (`id`)
) ENGINE=InnoDB AUTO_INCREMENT=5 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci

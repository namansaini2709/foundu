CREATE TABLE `freelancer_profiles` (
  `id` int NOT NULL AUTO_INCREMENT,
  `user_id` int NOT NULL,
  `freelancer_id` int NOT NULL,
  `first_name` varchar(100) DEFAULT NULL,
  `last_name` varchar(100) DEFAULT NULL,
  `email` varchar(255) DEFAULT NULL,
  `phone_number` varchar(20) DEFAULT NULL,
  `degree_type` varchar(100) DEFAULT NULL,
  `education_level` varchar(100) DEFAULT NULL,
  `graduation_year` varchar(10) DEFAULT NULL,
  `college` varchar(255) DEFAULT NULL,
  `university` varchar(255) DEFAULT NULL,
  `freelancer_type` varchar(100) DEFAULT NULL,
  `freelancer_type_other` varchar(255) DEFAULT NULL,
  `skills_description` text,
  `describe_best` text,
  `experience` varchar(100) DEFAULT NULL,
  `job_profile` varchar(255) DEFAULT NULL,
  `working_environment` varchar(255) DEFAULT NULL,
  `stipend_expectations` varchar(100) DEFAULT NULL,
  `stipend_other` varchar(100) DEFAULT NULL,
  `linkedin` varchar(255) DEFAULT NULL,
  `github` varchar(255) DEFAULT NULL,
  `other_links` text,
  `created_at` timestamp NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `user_id` (`user_id`),
  KEY `freelancer_id` (`freelancer_id`),
  CONSTRAINT `freelancer_profiles_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`id`),
  CONSTRAINT `freelancer_profiles_ibfk_2` FOREIGN KEY (`freelancer_id`) REFERENCES `users` (`id`)
) ENGINE=InnoDB AUTO_INCREMENT=6 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci

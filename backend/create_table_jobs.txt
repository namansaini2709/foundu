CREATE TABLE `foundu_jobs` (
  `job_id` int NOT NULL AUTO_INCREMENT,
  `user_id` int NOT NULL,
  `founder_id` int NOT NULL,
  `startup_name` varchar(255) NOT NULL,
  `job_title` varchar(255) NOT NULL,
  `freelancer_job_id` int DEFAULT NULL,
  `cofounder_job_id` int DEFAULT NULL,
  `job_type` enum('freelancer','cofounder') NOT NULL,
  `domain` varchar(100) DEFAULT NULL,
  `role_category` varchar(255) DEFAULT NULL,
  `work_type` varchar(100) DEFAULT NULL,
  `duration` varchar(100) DEFAULT NULL,
  `budget_range` varchar(100) DEFAULT NULL,
  `urgency` varchar(100) DEFAULT NULL,
  `job_level` varchar(50) DEFAULT NULL,
  `job_location_type` varchar(50) DEFAULT NULL,
  `job_requirements` text,
  `role_type` varchar(255) DEFAULT NULL,
  `compensation` varchar(100) DEFAULT NULL,
  `commitment` varchar(100) DEFAULT NULL,
  `priority_factors` text,
  `full_job_description` text NOT NULL,
  `created_at` timestamp NULL DEFAULT CURRENT_TIMESTAMP,
  `is_assigned` tinyint(1) DEFAULT '0',
  `payment_status` enum('pending','paid','failed') DEFAULT 'pending',
  `numeric_budget` decimal(10,2) DEFAULT '0.00',
  `gst_amount` decimal(10,2) DEFAULT '0.00',
  `commission_amount` decimal(10,2) DEFAULT '0.00',
  `total_amount` decimal(10,2) DEFAULT '0.00',
  `razorpay_order_id` varchar(255) DEFAULT NULL,
  `razorpay_payment_id` varchar(255) DEFAULT NULL,
  `is_completed` tinyint(1) DEFAULT '0',
  PRIMARY KEY (`job_id`),
  KEY `user_id` (`user_id`),
  KEY `founder_id` (`founder_id`),
  CONSTRAINT `foundu_jobs_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`id`),
  CONSTRAINT `foundu_jobs_ibfk_2` FOREIGN KEY (`founder_id`) REFERENCES `users` (`id`)
) ENGINE=InnoDB AUTO_INCREMENT=10007 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci

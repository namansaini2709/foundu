CREATE TABLE `startup_profiles` (
  `id` int NOT NULL AUTO_INCREMENT,
  `user_id` int NOT NULL,
  `founder_id` int NOT NULL,
  `startup_name` varchar(255) DEFAULT NULL,
  `startup_logo_url` text,
  `founder_type` varchar(100) DEFAULT NULL,
  `why_building` text,
  `superpower` varchar(100) DEFAULT NULL,
  `superpower_other` varchar(255) DEFAULT NULL,
  `weaknesses` json DEFAULT NULL,
  `hours_per_week` varchar(50) DEFAULT NULL,
  `cofounder_type` json DEFAULT NULL,
  `problem_solving` text,
  `who_has_problem` text,
  `problem_impact` varchar(100) DEFAULT NULL,
  `solution_building` text,
  `monetization` varchar(100) DEFAULT NULL,
  `unfair_advantage` text,
  `monthly_investment` varchar(100) DEFAULT NULL,
  `survival_without_revenue` varchar(100) DEFAULT NULL,
  `biggest_risk` text,
  `current_need` text,
  `outcome_30_days` text,
  `execution_failures` text,
  `trust_factors` json DEFAULT NULL,
  `use_vendors` varchar(50) DEFAULT NULL,
  `vendor_categories` json DEFAULT NULL,
  `vendor_preference` json DEFAULT NULL,
  `current_stage` varchar(100) DEFAULT NULL,
  `growth_blocker` text,
  `magic_fix` text,
  `dashboards` json DEFAULT NULL,
  `current_tools` text,
  `created_at` timestamp NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `user_id` (`user_id`),
  KEY `founder_id` (`founder_id`),
  CONSTRAINT `startup_profiles_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`id`),
  CONSTRAINT `startup_profiles_ibfk_2` FOREIGN KEY (`founder_id`) REFERENCES `users` (`id`)
) ENGINE=InnoDB AUTO_INCREMENT=4 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci
